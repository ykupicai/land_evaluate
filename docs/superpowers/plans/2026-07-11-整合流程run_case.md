# 整合流程 run_case 實作計畫

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 單一總控 `run_case.py`:使用者給 shp(trust-shp)或土地清冊 → 撈基地(清冊模式含比對+新舊地號 fallback 鏈)→ fishpv 臨路評估 → 00 總區位圖停點 → 六張基本圖(分區/用地/國土功能分區/空照/國土利用調查/臨路判定)+ .qgz。

**Architecture:** 子進程編排。run_case(系統 Python)以 subprocess 依序呼叫既有階段腳本;階段間以 `outputs/<案子>/` 檔案為契約。出圖(含新主題 5/6 與總區位圖)全部在 make_maps.py(PyQGIS 進程)內完成,維持單一 .qgz 寫出點。fishpv 以其 CLI 呼叫(QGIS 內建 Python,經 env.bat)。

**Tech Stack:** Python 3.14(系統)/PyQGIS 3.40.12/geopandas(fishpv 側)/openpyxl(僅系統 Python)/pytest。

**Spec:** `docs/superpowers/specs/2026-07-11-整合流程run_case-design.md`(必讀,含 CLI 介面 §3、資料流 §4、fallback 鏈 §5.2、六圖規格 §5.3)

```python
# -*- coding: utf-8 -*-
"""fallback 鏈寫檔端:把 04_新舊地號對照 回填進 01_中繼檔(spec §5.2)。

- 查詢狀態=成功:該列改用查得的段/小段/地號(宗地母/子重算),原值記備註。
- 查詢狀態=查無:該列標「比對模式=母號範圍」(match_parcels 會撈同段同宗地母全部宗地),
  地號原文改為母號,原值記備註。
用法:python -X utf8 scripts/apply_landno.py <案子名>
"""
import re
import sys
from pathlib import Path

REPO = Path(__file__).resolve().parent.parent
COLS_EXTRA = "比對模式"


def split_section(name: str) -> tuple[str, str]:
    """'聚興段新興小段'→('聚興','新興');'後壁段'→('後壁','');'民權一段'→('民權一','')。"""
    name = (name or "").strip()
    m = re.match(r"^(.*?)段(?:(.*?)小段)?$", name)
    if not m:
        return name, ""
    return m.group(1), (m.group(2) or "")


def _parse_no(s: str) -> tuple[str, str]:
    s = str(s).strip().split(".")[0]
    m = re.match(r"^(\d+)(?:-(\d+))?$", s)
    if not m:
        return "", ""
    return f"{int(m.group(1)):04d}", (f"{int(m.group(2)):04d}" if m.group(2) else "0000")


def apply_corrections(meta_rows: list[dict], lookup_rows: list[dict]):
    """依 04 修正 01 列。key=(案場ID, 原地號)。回傳 (新列list, 統計dict)。"""
    idx = {}
    for lr in lookup_rows:
        idx[(str(lr.get("案場ID", "")).strip().zfill(3),
             str(lr.get("原地號", "")).strip())] = lr
    stats = {"exact": 0, "mother": 0, "skip": 0}
    out = []
    for row in meta_rows:
        r = dict(row)
        key = (str(r.get("案場ID", "")).strip().zfill(3),
               str(r.get("地號原文", "")).strip())
        lr = idx.get(key)
        if lr is None:
            out.append(r)
            continue
        orig = f"{r['段']}{('段' if r['段'] else '')}{(str(r['小段']) + '小段') if str(r.get('小段') or '').strip() not in ('', 'None', 'nan') else ''}{r['地號原文']}"
        if str(lr.get("查詢狀態", "")).strip() == "成功":
            seg, sub = split_section(str(lr.get("查得新段名", "")))
            mother, child = _parse_no(lr.get("查得新地號", ""))
            if not seg or not mother:
                stats["skip"] += 1
                out.append(r)
                continue
            r["段"], r["小段"], r["地號原文"] = seg, sub, str(lr["查得新地號"]).strip()
            r["宗地母"], r["宗地子"] = mother, child
            r["備註"] = (str(r.get("備註") or "") +
                         f"｜原{orig},依新舊地號對照回填").lstrip("｜")
            stats["exact"] += 1
        else:  # 查無 → 母號範圍
            mother = str(r.get("宗地母", "")).strip().split(".")[0]
            if not mother:
                stats["skip"] += 1
                out.append(r)
                continue
            r[COLS_EXTRA] = "母號範圍"
            r["地號原文"] = str(int(mother))
            r["宗地母"], r["宗地子"] = mother.zfill(4), ""
            r["備註"] = (str(r.get("備註") or "") +
                         f"｜原{orig}查無對應,以母號範圍示意出圖").lstrip("｜")
            stats["mother"] += 1
        out.append(r)
    return out, stats


def main():
    from openpyxl import Workbook, load_workbook
    case = sys.argv[1]
    d = REPO / "outputs" / case
    def read(p):
        ws = load_workbook(p, read_only=True, data_only=True).active
        rows = list(ws.iter_rows(values_only=True))
        hdr = [str(c) for c in rows[0]]
        return [dict(zip(hdr, [("" if v is None else v) for v in row]))
                for row in rows[1:] if any(v is not None for v in row)], hdr
    meta, hdr = read(d / "01_中繼檔.xlsx")
    lookup, _ = read(d / "04_新舊地號對照清單.xlsx")
    out, stats = apply_corrections(meta, lookup)
    if COLS_EXTRA not in hdr:
        hdr = hdr + [COLS_EXTRA]
    wb = Workbook(); ws = wb.active
    ws.append(hdr)
    for r in out:
        ws.append([r.get(h, "") for h in hdr])
    wb.save(d / "01_中繼檔.xlsx")
    print(f"回填完成:精確 {stats['exact']}、母號範圍 {stats['mother']}、略過 {stats['skip']}")


if __name__ == "__main__":
    main()
```

```python
def load_sitemap(path: str) -> dict:
    """sitemap.csv(地號,案場ID,案場名稱)→ {地號:(案場ID,案場名稱)}。"""
    import csv
    out = {}
    with open(path, encoding="utf-8-sig", newline="") as f:
        for row in csv.DictReader(f):
            no = str(row.get("地號", "")).strip()
            if no:
                out[no] = (str(row.get("案場ID", "")).strip(),
                           str(row.get("案場名稱", "")).strip())
    return out
```

```python
def luimap_legend_categories(features_geojson: Path, colortable_csv: Path,
                             extent: QgsRectangle) -> list:
    """圖面範圍內實際出現的 LUIMAP 官方類別(依 features 面積佔比)。
    回傳 [(lu_name, (r,g,b)), ...] 佔比由大到小,<LUIMAP_SHARE_MIN 不列。"""
    import csv as _csv
    import json as _json
    colors = {}
    with open(colortable_csv, encoding="utf-8-sig", newline="") as f:
        for row in _csv.DictReader(f):
            colors[str(row["lu_code"]).strip()] = (
                row["lu_name"].strip(),
                (int(row["R"]), int(row["G"]), int(row["B"])))
    lyr = QgsVectorLayer(str(features_geojson), "f", "ogr")
    if not lyr.isValid():
        return []
    req = QgsFeatureRequest().setFilterRect(extent)
    shares = defaultdict(float)
    total = 0.0
    for f in lyr.getFeatures(req):
        code = attr_str(f["lu_code"])
        a = f.geometry().area()
        shares[code] += a
        total += a
    if total <= 0:
        return []
    out = []
    for code, a in sorted(shares.items(), key=lambda x: -x[1]):
        if a / total < LUIMAP_SHARE_MIN or code not in colors:
            continue
        out.append(colors[code])
    return out
```

```python
def road_status_color(status: str) -> tuple:
    s = str(status or "")
    if "不臨路" in s:
        return (200, 0, 0)
    if "直接臨路" in s:
        return (0, 150, 0)
    if "需搭橋" in s:
        return (255, 140, 0)
    return (120, 120, 120)


def load_sites_status(csv_path: Path) -> dict:
    """sites_status.csv → {site_id: (臨路狀態, 距離str, 註記)}。"""
    import csv as _csv
    out = {}
    if not csv_path.exists():
        return out
    with open(csv_path, encoding="utf-8-sig", newline="") as f:
        for row in _csv.DictReader(f):
            out[str(row["site_id"]).strip()] = (
                str(row.get("臨路狀態", "")), str(row.get("臨路距離_m", "")),
                str(row.get("臨路註記", "")))
    return out
```

```python
def plan_steps(mode_shp: bool, resume: bool, chosen_mode, artifacts: dict) -> list:

        case_dir / "臨路評估" / "sites_status.csv", index=False, encoding="utf-8-sig")
    rows = []
    for _, r in df.iterrows():
        for no in str(r["地號清單"]).split(";"):
            no = no.strip()
            if no:
                rows.append({"地號": no, "案場ID": r["site_id"],
                             "案場名稱": f"{r['site_id']}宗基地"})
    pd.DataFrame(rows).to_csv(case_dir / "臨路評估" / "sitemap.csv",
                              index=False, encoding="utf-8-sig")
    print(f"digest:sites_status.csv + sitemap.csv({len(rows)} 地號)")


def _run(cmd, cwd=None, label=""):
    print(f"\n=== {label or ' '.join(map(str, cmd))} ===", flush=True)
    r = subprocess.run([str(c) for c in cmd], cwd=cwd)
    if r.returncode not in (0, 5):  # 5=QGIS 慣有結束碼
        sys.exit(f"[run_case] 步驟失敗({label}),returncode={r.returncode}")


def main():
    args = sys.argv[1:]
    shp = lst = case = chosen = None
    resume = False
    passthru = []
    it = iter(range(len(args)))
    i = 0
    while i < len(args):
        a = args[i]
        if a == "--shp":
            shp = args[i + 1]; i += 2
        elif a == "--list":
            lst = args[i + 1]; i += 2
        elif a == "--case":
            case = args[i + 1]; i += 2
        elif a == "--mode":
            chosen = args[i + 1]; i += 2
        elif a == "--resume":
            resume = True; i += 1
        elif a in ("--no-qgz", "--mem", "--basic"):
            passthru.append(a); i += 1
        else:
            sys.exit(f"未知參數 {a}")
    if not (shp or lst):
        sys.exit("需 --shp <基地.shp> 或 --list <清冊>")
    if chosen not in (None, "single", "persite"):
        sys.exit("--mode 需為 single|persite")
    mode_shp = shp is not None
    if not case:
        case = Path(shp or lst).stem
    case_dir = REPO / "outputs" / case
    case_dir.mkdir(parents=True, exist_ok=True)
    fp = case_dir / "臨路評估"

    def art():
        n03 = case_dir / "03_無法對應清單.xlsx"
        nonempty = False
        if n03.exists():
            import pandas as pd
            nonempty = len(pd.read_excel(n03)) > 0
        return {"01": (case_dir / "01_中繼檔.xlsx").exists(),
                "02": (case_dir / "02_比對結果.xlsx").exists(),
                "03_nonempty": nonempty,
                "fishpv": (fp / "成果.xlsx").exists(),
                "overview": (case_dir / "00_總區位圖.png").exists()}

    dbf = str(Path(shp).with_suffix(".dbf")) if shp else None
    steps = plan_steps(mode_shp, resume, chosen, art())
    print(f"[run_case] 案子={case} 步驟={steps}")
    for s in steps:
        if s == "meta":
            _run(["python", "-X", "utf8", REPO / "scripts" / "shp_to_meta.py",
                  dbf, case, case], label="shp_to_meta")
        elif s == "meta_sitemap":
            _run(["python", "-X", "utf8", REPO / "scripts" / "shp_to_meta.py",
                  dbf, case, case, "--sitemap", fp / "sitemap.csv"],
                 label="shp_to_meta --sitemap")
        elif s == "parse":
            sys.exit("[run_case] 清冊解析請先以既有流程產 01_中繼檔(換版式需 AI 寫解析),"
                     "完成人工核對後加 --resume 續跑")
        elif s in ("match", "match2"):
            _run(["python", "-X", "utf8", REPO / "scripts" / "match_parcels.py", case],
                 label=s)
            if s == "match" and "landno" not in steps and art()["03_nonempty"]:
                # 首次比對後才知道 03 是否非空 → 動態補跑 fallback
                for extra in ("landno", "apply", "match2"):
                    steps.append(extra) if extra not in steps else None
        elif s == "landno":
            _run(["python", "-X", "utf8", REPO / "scripts" / "query_landno.py", case],
                 label="query_landno")
        elif s == "apply":
            _run(["python", "-X", "utf8", REPO / "scripts" / "apply_landno.py", case],
                 label="apply_landno")
        elif s == "extract":
            _run([QGIS_BAT, REPO / "scripts" / "make_maps.py", case, "--extract-only"],
                 label="extract gpkg")
        elif s == "fishpv":
            cad = shp if mode_shp else str(case_dir / "圖層.gpkg")
            _run([FISHPV_ROOT / "env.bat", "python", "-m", "fishpv.cli",
                  "--cadastre", cad, "--colortable", COLORTABLE,
                  "--out", fp, "--no-ai", "--landno-field", "地號"],
                 cwd=str(FISHPV_ROOT), label="fishpv 臨路評估")
        elif s == "digest":
            digest_fishpv(case_dir)
        elif s == "overview":
            _run([QGIS_BAT, REPO / "scripts" / "make_maps.py", case, "--overview-only"],
                 label="00_總區位圖")
        elif s == "maps":
            _run([QGIS_BAT, REPO / "scripts" / "make_maps.py", case, *passthru],
                 label="六張圖+qgz")
        elif s == "STOP_MODE":
            print(f"\n✋ 請看 {case_dir / '00_總區位圖.png'} 後選出圖單位,"
                  f"再跑同指令加 --mode single|persite")
            return
        elif s == "STOP_CHECK":
            return
    print("\n[run_case] 完成")


if __name__ == "__main__":
    main()
```
